--3. XPath Expressions
--Query 1 (Returns all product elements directly under catalog.)

SELECT id, xDoc.query('/catalog/product')
FROM AdminDocs;

--Query 2 (Returns all product nodes anywhere in the document.)

SELECT id, xDoc.query('//product')
FROM AdminDocs;

--Query 3 (Selects product elements under the root node.)

SELECT id, xDoc.query('/*/product')
FROM AdminDocs;

--Query 4 (Products with department WMN)

SELECT id, xDoc.query('/*/product[@dept="WMN"]')
FROM AdminDocs;

--Query 5 (Using child and attribute axis)

SELECT id, xDoc.query('/*/child::product[attribute::dept="WMN"]')
FROM AdminDocs;

--Query 6 (Find product where dept = WMN)

SELECT id, xDoc.query('//product[@dept="WMN"]')
FROM AdminDocs;

--Query 7 (Descendant search)

SELECT id, xDoc.query('descendant-or-self::product[attribute::dept="WMN"]')
FROM AdminDocs;

--Query 8 (Products with number > 500)
SELECT id, xDoc.query('//product[number > 500]')
FROM AdminDocs
WHERE id = 1;

--Query 9 (Product numbers > 500)
SELECT id, xDoc.query('//product/number[. > 500]')
FROM AdminDocs
WHERE id = 1;

--Query 10 (4'th product)

SELECT id, xDoc.query('/catalog/product[4]')
FROM AdminDocs
WHERE id = 1;


--Query 11 

SELECT id, xDoc.query('//product[number > 500][@dept="ACC"]')
FROM AdminDocs
WHERE id = 1;


----Query 12 

SELECT id, xDoc.query('//product[number > 500][1]')
FROM AdminDocs
WHERE id = 1;


--4. XQuery Expressions
--Query 1 (Return product numbers)

SELECT xDoc.query('
for $prod in //product
let $x := $prod/number
return $x')
FROM AdminDocs
WHERE id = 1;


--Query 2 (Product numbers greater than 500)

SELECT xDoc.query('
for $prod in //product
let $x := $prod/number
where $x > 500
return $x')
FROM AdminDocs
WHERE id = 1;


--Query 3 (Same as Query 1)

SELECT xDoc.query('
for $prod in //product
let $x := $prod/number
return $x')
FROM AdminDocs
WHERE id = 1;

--Query 4 (Return numbers as XML item)

SELECT xDoc.query('
for $prod in //product
let $x := $prod/number
where $x > 500
return (<Item>{$x}</Item>)')
FROM AdminDocs
WHERE id = 1;

--Query 5 (Using condition inside loop)

SELECT xDoc.query('
for $prod in //product[number > 500]
let $x := $prod/number
return (<Item>{$x}</Item>)')
FROM AdminDocs
WHERE id = 1;

--Query 6 (Using data() function)

SELECT xDoc.query('
for $prod in //product
let $x := $prod/number
where $x > 500
return (<Item>{data($x)}</Item>)')
FROM AdminDocs
WHERE id = 1;

--Query 7 (If-Else condition)

SELECT xDoc.query('
for $prod in //product
let $x := $prod/number
return
if ($x > 500)
then <book>{data($x)}</book>
else <paper>{data($x)}</paper>')
FROM AdminDocs
WHERE id = 1;



--5. XML DML Operations
--Step 1 – View Record

SELECT * FROM AdminDocs WHERE id = 2;


--Step 2 – Insert New Section

UPDATE AdminDocs
SET xDoc.modify('
insert
<section num="2">
<title>Background</title>
</section>
after (/doc//section[@num=1])[1]');


--Step 3 – Delete Section

UPDATE AdminDocs
SET xDoc.modify('
delete
//section[@num="2"]')
