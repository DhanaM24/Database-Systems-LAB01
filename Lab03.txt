-- Address object
CREATE OR REPLACE TYPE address_type AS OBJECT (
    street_number NUMBER,
    street_name VARCHAR2(50),
    suburb VARCHAR2(50),
    state VARCHAR2(20),
    pin VARCHAR2(10)
);
/

-- Investment object
CREATE OR REPLACE TYPE investment_type AS OBJECT (
    company VARCHAR2(50),
    purchase_price NUMBER(10,2),
    purchase_date DATE,
    qty NUMBER
);
/

-- Nested table for investments
CREATE OR REPLACE TYPE investment_table AS TABLE OF investment_type;
/

-- VARRAY for exchanges
CREATE OR REPLACE TYPE exchange_array AS VARRAY(10) OF VARCHAR2(50);
/

-- Client object
CREATE OR REPLACE TYPE client_type AS OBJECT (
    firstname VARCHAR2(50),
    lastname VARCHAR2(50),
    address address_type,
    investments investment_table
);
/

-- Stock object
CREATE OR REPLACE TYPE stock_type AS OBJECT (
    company VARCHAR2(50),
    current_price NUMBER(10,2),
    exchanges exchange_array,
    last_dividend NUMBER(10,2),
    eps NUMBER(10,2)
);
/

-- Create Tables
CREATE TABLE clients_obj OF client_type (
    PRIMARY KEY (firstname, lastname)
)
NESTED TABLE investments STORE AS investment_store;

CREATE TABLE stocks_obj OF stock_type (
    company PRIMARY KEY
);

-- 4. QUERIES

-- (a)
SELECT c.firstname, c.lastname,
       i.company, s.current_price, s.last_dividend, s.eps
FROM clients_obj c,
     TABLE(c.investments) i,
     stocks_obj s
WHERE i.company = s.company;

-- (b)
SELECT c.firstname, c.lastname,
       i.company,
       SUM(i.qty) total_qty,
       ROUND(SUM(i.qty*i.purchase_price)/SUM(i.qty),2) avg_price
FROM clients_obj c,
     TABLE(c.investments) i
GROUP BY c.firstname, c.lastname, i.company;

-- (c)
SELECT s.company,
       c.firstname,
       c.lastname,
       SUM(i.qty) qty,
       SUM(i.qty)*s.current_price AS current_value
FROM clients_obj c,
     TABLE(c.investments) i,
     stocks_obj s,
     TABLE(s.exchanges) e
WHERE i.company = s.company
AND e.COLUMN_VALUE='New York'
GROUP BY s.company, c.firstname, c.lastname, s.current_price;

-- (d)
SELECT c.firstname, c.lastname,
       SUM(i.qty*i.purchase_price) total_purchase_value
FROM clients_obj c,
     TABLE(c.investments) i
GROUP BY c.firstname, c.lastname;

-- (e)
SELECT c.firstname, c.lastname,
       SUM(i.qty*s.current_price) - SUM(i.qty*i.purchase_price) AS book_profit
FROM clients_obj c,
     TABLE(c.investments) i,
     stocks_obj s
WHERE i.company = s.company
GROUP BY c.firstname, c.lastname;
