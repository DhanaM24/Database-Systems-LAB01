ALTER TYPE client_t ADD MEMBER FUNCTION total_purchase_value RETURN NUMBER CASCADE;
/
CREATE OR REPLACE TYPE BODY client_t AS
  MEMBER FUNCTION total_purchase_value RETURN NUMBER IS
    v_total NUMBER := 0;
  BEGIN
    IF investments IS NULL THEN
      RETURN 0;
    END IF;

    FOR i IN 1 .. investments.COUNT LOOP
      IF investments(i).purchase_price IS NOT NULL AND investments(i).qty IS NOT NULL THEN
        v_total := v_total + (investments(i).purchase_price * investments(i).qty);
      END IF;
    END LOOP;

    RETURN v_total;
  END total_purchase_value;
END;
/

CREATE OR REPLACE TYPE BODY client_t AS
  MEMBER FUNCTION total_purchase_value RETURN NUMBER IS
    v_total NUMBER := 0;
  BEGIN
    IF investments IS NULL THEN
      RETURN 0;
    END IF;

    FOR i IN 1 .. investments.COUNT LOOP
      IF investments(i).purchase_price IS NOT NULL AND investments(i).qty IS NOT NULL THEN
        v_total := v_total + (investments(i).purchase_price * investments(i).qty);
      END IF;
    END LOOP;

    RETURN v_total;
  END total_purchase_value;

  MEMBER FUNCTION total_profit RETURN NUMBER IS
    v_total NUMBER := 0;
    v_obj   stock_t;
  BEGIN
    IF investments IS NULL THEN
      RETURN 0;
    END IF;

    FOR i IN 1 .. investments.COUNT LOOP
      IF investments(i).stock_ref IS NOT NULL
         AND investments(i).purchase_price IS NOT NULL
         AND investments(i).qty IS NOT NULL THEN
        v_obj := DEREF(investments(i).stock_ref);
        IF v_obj.current_price IS NOT NULL THEN
          v_total := v_total + ((v_obj.current_price - investments(i).purchase_price) * investments(i).qty);
        END IF;
      END IF;
    END LOOP;

    RETURN v_total;
  END total_profit;
END;

  s.current_price,
  s.exchange_count() AS no_of_exchanges
FROM stocks s
WHERE s.exchange_count() > 1;

SELECT
  c.firstname || ' ' || c.lastname AS client_name,
  DEREF(i.stock_ref).company AS stock_name,
  DEREF(i.stock_ref).yield_pct() AS yield,
  DEREF(i.stock_ref).current_price AS current_price,
  DEREF(i.stock_ref).earnings_per_share AS earnings_per_share
FROM clients c,
     TABLE(c.investments) i
ORDER BY client_name, stock_name;
